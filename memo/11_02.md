philo_eat():
- ph->state = PH_EATING
- フォークを取る

- フォークが取れなかったら(死亡した or 哲学者の数が一人)、終了フラグをオンにする（その後の処理は実行しない）
- 哲学者のmutexを取得して、eat_count と last_eat_at を更新する (その後ロック解除)
- 次食べる時間を設定する
- time_to_eat間眠る

- フォークを解除

static bool philo_eat(t_philo *philo) {
  philo->state = PH_EATING;
  is_end =  pickup forks
  if (!is_end) {
    lock philo mutex (eat_count and last_eat_at)
    eat_count++
    wirte log and get last_eat_at
    philo->next_eat_at = add(start + interval)
    sleep_until(last_eat_at + time_to_sleep)
    unlock philo mutex
  }
  unlock forks
  return is_end
}

usleep_since(t, interval)
usleep_until(timeadd(t, interval))

pickup_forks() {
  is_end = false
  if (first->id == second->id)
    return true
  lock(first)
  is_end = write_log()
  if (is_end)
    unlock(first)
  else {
    lock(second)
    is_end = write_log()
    if (is_end)
      unlock(second)
  }
  return is_end;
}

- もし哲学者が死んでいたら、true を返す (ちなみに満腹の場合はチェック済み)
write_log(t_philo *philo, message, t_timeval *tp) {
  bool is_end;
  lock(monitor.mutex)
  t_timeval now;
  is_end = unsafe_is_dead(*philo, &now)
  if (is_end)
    unsafe_write_dead_log(now)
  else {
    print message
  }
  if (tp != NULL)
    *tp = now
  unlock(monitor.mutex)
}
